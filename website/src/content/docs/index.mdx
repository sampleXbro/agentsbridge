  title="Agents that learn from your repo"
  lead="A shared, git-tracked memory. Recall before every edit, capture after every failure. Same mistake, never twice, in any tool."
>
  <LessonsLoop items={HOME_LESSON_LOOP} links={HOME_LESSON_LINKS}>

```bash
agentsmesh init --lessons && agentsmesh generate

# after a failure, the agent captures a rule
agentsmesh lessons add "Normalize CLI display paths to forward slashes" \
  --topic windows-paths --trigger-file "src/cli/**/*.ts"

# before the next edit, it recalls the rule
agentsmesh lessons query --file src/cli/foo.ts
#   -> Normalize CLI display paths to forward slashes
```

  </LessonsLoop>
</Section>
